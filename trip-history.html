import { auth, db, logoutUser } from "./firebase.js"; import { onAuthStateChanged } from "https://www.gstatic.com/firebasejs/13.1.0/firebase-auth.js"; import { collection, query, where, getDocs, doc, updateDoc, serverTimestamp } from "https://www.gstatic.com/firebasejs/13.1.0/firebase-firestore.js"; const historyList = document.getElementById("tripHistoryList"); const messageBox = document.getElementById("historyMessage"); const totalTripsElement = document.getElementById("totalTrips"); const completedTripsElement = document.getElementById("completedTrips"); const outstandingBalanceElement = document.getElementById("outstandingBalance"); let currentUser = null; let trips = []; let payments = []; let activeFilter = "all"; function showMessage(message, type = "error") { messageBox.textContent = message; messageBox.className = `message show ${type}`; } function clearMessage() { messageBox.textContent = ""; messageBox.className = "message"; } function createElement(tag, className, text) { const element = document.createElement(tag); if (className) element.className = className; if (text !== undefined) element.textContent = text; return element; } function formatDate(value) { if (!value) return "Date unavailable"; if (value.toDate) { value = value.toDate(); } else { value = new Date(value); } if (Number.isNaN(value.getTime())) { return "Date unavailable"; } return value.toLocaleDateString("en-IN", { day: "numeric", month: "short", year: "numeric" }); } function formatMoney(value) { return new Intl.NumberFormat("en-IN", { style: "currency", currency: "INR", maximumFractionDigits: 0 }).format(Number(value) || 0); } function getPayment(tripId) { return payments.find(payment => payment.id === tripId) || null; } function isArchived(trip) { return trip.archived === true || trip.status === "archived"; } function getStatusClass(status) { const allowed = [ "completed", "archived", "cancelled", "assigned", "open", "in_progress" ]; return allowed.includes(status) ? `status-${status}` : ""; } function getStatusLabel(trip) { if (isArchived(trip)) return "Archived"; const labels = { completed: "Completed", cancelled: "Cancelled", assigned: "Assigned", open: "Open", in_progress: "In Progress" }; return labels[trip.status] || "Unknown"; } function isActiveTrip(trip) { return ["open", "assigned", "in_progress"].includes(trip.status); } function getFilteredTrips() { if (activeFilter === "completed") { return trips.filter( trip => trip.status === "completed" && !isArchived(trip) ); } if (activeFilter === "active") { return trips.filter(isActiveTrip); } if (activeFilter === "archived") { return trips.filter(isArchived); } return trips; } function updateSummary() { const completed = trips.filter( trip => trip.status === "completed" && !isArchived(trip) ); const outstanding = payments.reduce((total, payment) => { const agreed = Number(payment.agreedFare) || 0; const paid = Number(payment.amountPaid) || 0; return total + Math.max(0, agreed - paid); }, 0); totalTripsElement.textContent = trips.length; completedTripsElement.textContent = completed.length; outstandingBalanceElement.textContent = formatMoney(outstanding); } function renderTrips() { historyList.replaceChildren(); const filtered = getFilteredTrips(); if (filtered.length === 0) { const empty = createElement("div", "empty-state"); empty.appendChild( createElement("strong", "", "No trips found") ); empty.appendChild( createElement( "p", "", "Your trip records will appear here when you post or accept trips." ) ); historyList.appendChild(empty); return; } filtered.forEach(trip => { historyList.appendChild(createTripCard(trip)); }); } function createTripCard(trip) { const card = createElement("article", "trip-card"); const top = createElement("div", "trip-top"); const details = createElement("div"); const route = createElement( "h3", "route", `${trip.source || "Unknown"} → ${trip.destination || "Unknown"}` ); const status = createElement( "span", `status ${getStatusClass(isArchived(trip) ? "archived" : trip.status)}`, getStatusLabel(trip) ); const meta = createElement("div", "trip-meta"); const date = createElement( "span", "", `Travel date: ${trip.travelDate || "Not specified"}` ); const time = createElement( "span", "", `Pickup: ${trip.travelTime || "Not specified"}` ); const type = createElement( "span", "", `Trip type: ${trip.tripType || "Not specified"}` ); meta.append(date, time, type); details.append(route, meta); top.append(details, status); card.appendChild(top); const payment = getPayment(trip.id); const agreedFare = payment ? Number(payment.agreedFare) || 0 : Number(trip.fare) || 0; const amountPaid = payment ? Number(payment.amountPaid) || 0 : 0; const outstanding = Math.max(0, agreedFare - amountPaid); const paymentRow = createElement("div", "payment-row"); const fare = createElement("div"); fare.append( createElement("span", "muted", "Agreed fare"), createElement("br"), createElement("strong", "", formatMoney(agreedFare)) ); const paid = createElement("div"); paid.append( createElement("span", "muted", "Amount paid"), createElement("br"), createElement("strong", "", formatMoney(amountPaid)) ); const balance = createElement("div"); balance.append( createElement("span", "muted", "Outstanding"), createElement("br"), createElement("strong", "", formatMoney(outstanding)) ); paymentRow.append(fare, paid, balance); card.appendChild(paymentRow); const paymentStatus = payment?.paymentStatus || "unpaid"; const paymentInfo = createElement( "p", "muted", `Payment status: ${paymentStatus.replaceAll("_", " ")}` ); paymentInfo.style.margin = "12px 0 0"; card.appendChild(paymentInfo); const actions = createElement("div", "actions"); if ( trip.posterUid === currentUser.uid && trip.status === "completed" ) { const archiveButton = createElement( "button", "btn btn-secondary", isArchived(trip) ? "Restore Trip" : "Archive Trip" ); archiveButton.type = "button"; archiveButton.addEventListener("click", () => { toggleArchive(trip, archiveButton); }); actions.appendChild(archiveButton); } if (trip.status === "assigned" || trip.status === "in_progress") { const info = createElement( "span", "muted", "Completion requires confirmation from both parties." ); actions.appendChild(info); } if (actions.children.length > 0) { card.appendChild(actions); } return card; } async function toggleArchive(trip, button) { if (!currentUser || trip.posterUid !== currentUser.uid) { showMessage("You are not authorised to change this trip."); return; } if (trip.status !== "completed") { showMessage("Only completed trips can be archived or restored."); return; } const currentlyArchived = isArchived(trip); const confirmed = window.confirm( currentlyArchived ? "Restore this completed trip to your history?" : "Archive this completed trip? You can restore it later." ); if (!confirmed) return; button.disabled = true; button.textContent = "Updating..."; try { await updateDoc( doc(db, "trips", trip.id), { archived: !currentlyArchived, archivedAt: currentlyArchived ? null : serverTimestamp(), updatedAt: serverTimestamp() } ); trip.archived = !currentlyArchived; if (currentlyArchived) { delete trip.archivedAt; } else { trip.archivedAt = new Date(); } clearMessage(); showMessage( currentlyArchived ? "Trip restored successfully." : "Trip archived successfully.", "success" ); updateSummary(); renderTrips(); } catch (error) { console.error("Trip archive error:", error); showMessage( error.code === "permission-denied" ? "Firebase rules did not allow this operation. Check that the trip is completed and belongs to you." : "Unable to update this trip. Please try again." ); button.disabled = false; button.textContent = currentlyArchived ? "Restore Trip" : "Archive Trip"; } } async function loadTrips(user) { historyList.replaceChildren(); historyList.appendChild( createElement("div", "empty-state", "Loading your trip records...") ); try { const tripsCollection = collection(db, "trips"); const postedQuery = query( tripsCollection, where("posterUid", "==", user.uid) ); const assignedQuery = query( tripsCollection, where("assignedDriverUid", "==", user.uid) ); const [postedSnapshot, assignedSnapshot] = await Promise.all([ getDocs(postedQuery), getDocs(assignedQuery) ]); const tripMap = new Map(); postedSnapshot.docs.forEach(item => { tripMap.set(item.id, { id: item.id, ...item.data() }); }); assignedSnapshot.docs.forEach(item => { tripMap.set(item.id, { id: item.id, ...item.data() }); }); trips = Array.from(tripMap.values()); trips.sort((a, b) => { const dateA = a.createdAt?.toMillis ? a.createdAt.toMillis() : 0; const dateB = b.createdAt?.toMillis ? b.createdAt.toMillis() : 0; return dateB - dateA; }); await loadPayments(user); updateSummary(); renderTrips(); } catch (error) { console.error("Trip history loading error:", error); historyList.replaceChildren(); historyList.appendChild( createElement( "div", "empty-state", "Unable to load your trip history. Please refresh the page." ) ); showMessage( error.code === "permission-denied" ? "Firestore denied access. Check your trip and payment security rules." : "There was a problem loading your trip history." ); } } async function loadPayments(user) { const paymentCollection = collection(db, "paymentRecords"); const posterQuery = query( paymentCollection, where("posterUid", "==", user.uid) ); const driverQuery = query( paymentCollection, where("driverUid", "==", user.uid) ); const [posterSnapshot, driverSnapshot] = await Promise.all([ getDocs(posterQuery), getDocs(driverQuery) ]); const paymentMap = new Map(); posterSnapshot.docs.forEach(item => { paymentMap.set(item.id, { id: item.id, ...item.data() }); }); driverSnapshot.docs.forEach(item => { paymentMap.set(item.id, { id: item.id, ...item.data() }); }); payments = Array.from(paymentMap.values()); } document.querySelectorAll("[data-filter]").forEach(button => { button.addEventListener("click", () => { activeFilter = button.dataset.filter; document.querySelectorAll("[data-filter]").forEach(filter => { filter.classList.toggle("active", filter === button); }); renderTrips(); }); }); document.getElementById("logoutBtn").addEventListener( "click", async () => { try { await logoutUser(); window.location.replace("login.html"); } catch (error) { console.error("Logout error:", error); showMessage("Unable to log out. Please try again."); } } ); onAuthStateChanged(auth, async user => { currentUser = user; if (!user) { window.location.replace("login.html"); return; } await loadTrips(user); });
